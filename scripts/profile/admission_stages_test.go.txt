func TestAdmissionPrivateProfileActualSQLitePragmasAndQueryPlans(t *testing.T) {

  {"candidateParticipantTrigger",`SELECT 1 FROM participants WHERE collection_id=? AND id=? AND included=1`,[]any{"profile-unused","profile-unused"}},
  {"candidateWinnerTrigger",`SELECT 1 FROM winners WHERE collection_id=? AND participant_id=?`,[]any{"profile-unused","profile-unused"}},

  {"operationLookup",`SELECT operation_id,kind,collection_id,round_id,status,revision,public_fingerprint,failure_code FROM operations WHERE operation_id=?`,[]any{"profile-unused"}},
  {"timerHint",`SELECT EXISTS(SELECT 1 FROM rounds WHERE state IN ('scheduled','executing'))`,nil},

}

func TestAdmissionPrivateManualPagedRerunReplayRollbackAndReopenTraceExactWithObserverOffOn(t *testing.T) {
 oldRecord:=profileAdmissionRecord
 defer func(){profileAdmissionRecord=oldRecord}()
 var originalTrace, originalRows string
 for _,observed:=range []bool{false,true} {
  store:=migratedTestStore(t)
  entropy,kdf:=new(productEntropy),new(productKDF)
  service:=productService(t,store,kdf,entropy,func()time.Time{return storageNow},"session-one",nil)
  ctx:=context.Background();finish:=func(){}
  if observed {ctx,finish=producttrace.Root(ctx,"test.privateManualPagedTrace")}
  request:=productRequest(storageNow,rl.ImmediateMode)
  request.Collection.Participants[0].Manual=&rl.ManualStateSnapshot{ManualIncluded:true}
  request.Collection.Participants[1].Manual=&rl.ManualStateSnapshot{OverrideExcluded:true}
  var rounds []rl.RoundRecord
  round,err:=service.CreateCollection(ctx,request);if err!=nil{t.Fatal(err)}
  rounds=append(rounds,round)
  for number:=2;number<=3;number++ {
   rerun:=rl.RerunRequest{OperationID:contracts.OperationID(fmt.Sprintf("private-paged-%d",number)),Context:contracts.CollectionContext{BackendSessionID:"session-one",CollectionID:round.CollectionID,Revision:round.Revision},Prizes:[]rl.Prize{{ID:"private-paged-prize",Count:1}},Mode:rl.ImmediateMode}
   round,err=service.Rerun(ctx,rerun);if err!=nil{t.Fatal(err)}
   replay,replayErr:=service.Rerun(ctx,rerun)
   if replayErr!=nil || !reflect.DeepEqual(round,replay){t.Fatal("observer changed rerun replay",replayErr)}
   rounds=append(rounds,round)
  }
  var pages []rl.CollectionPage
  for _,query:=range []rl.CollectionPageQuery{{Limit:1},{Limit:1,Offset:2},{Limit:1,RoundID:rounds[0].ID},{Limit:50,Offset:^uint32(0)}} {
   page,readErr:=store.ReadCollectionPage(ctx,round.CollectionID,query)
   if readErr!=nil || page.Total!=3 || page.ConsumedCount!=3 || page.LatestRound.ID!=round.ID || page.Revision!=round.Revision {t.Fatal("observer changed atomic latest/global page",readErr)}
   if page.Frozen.Participants[0].Manual==nil || !page.Frozen.Participants[0].Manual.ManualIncluded || page.Frozen.Participants[1].Manual==nil || !page.Frozen.Participants[1].Manual.OverrideExcluded || page.Frozen.Participants[2].Manual!=nil {t.Fatal("observer changed explicit manual bits/legacy unknown")}
   pages=append(pages,page)
  }
  durable:=corruptionDurableSnapshot(t,store)
  // The second candidate fails only after participant/operation/round and first
  // candidate writes, exercising the original outer rollback and close paths.
  execSQL(t,store.db,`CREATE TRIGGER private_admission_fault BEFORE INSERT ON round_candidates WHEN NEW.position=1 BEGIN SELECT RAISE(ABORT,'injected profile admission failure'); END`)
  failing:=productRequest(storageNow,rl.ImmediateMode)
  failing.OperationID="private-paged-failure"
  failing.Context.DraftID="private-paged-failure-draft"
  failing.Collection.SourceDraftID=failing.Context.DraftID
  _,failErr:=service.CreateCollection(ctx,failing)
  requireFault(t,failErr,contracts.StorageUnavailable)
  if durable!=corruptionDurableSnapshot(t,store) || entropy.calls.Load()!=3 || kdf.derive.Load()!=2 {t.Fatal("observer changed partial rollback or KDF/RNG boundary")}
  operation,readErr:=store.ReadOperation(ctx,failing.OperationID)
  if readErr!=nil || operation.Status!=contracts.OperationUnknown {t.Fatal("observer persisted failed admission",readErr)}
  execSQL(t,store.db,"DROP TRIGGER private_admission_fault")
  assertRoundSQLIntegrity(t,store);assertNoHandles(t,store)
  if err=service.Close(context.Background());err!=nil{t.Fatal(err)}
  path:=store.path
  if err=store.Close();err!=nil{t.Fatal(err)}
  reopened,err:=Open(context.Background(),path);if err!=nil{t.Fatal(err)}
  restored,err:=reopened.ReadCollectionPage(ctx,round.CollectionID,rl.CollectionPageQuery{Limit:1,Offset:2})
  if err!=nil || !reflect.DeepEqual(restored,pages[1]) || durable!=corruptionDurableSnapshot(t,reopened){reopened.Close();t.Fatal("observer changed reopened manual/page graph",err)}
  assertNoHandles(t,reopened)
  rows:=corruptionDurableSnapshot(t,reopened)
  if err=reopened.Close();err!=nil{t.Fatal(err)}
  trace:=struct{Rounds []rl.RoundRecord;Pages []rl.CollectionPage;Restored rl.CollectionPage;Unknown rl.OperationRecord;Fault contracts.ErrorCode}{rounds,pages,restored,operation,contracts.StorageUnavailable}
  raw,err:=json.Marshal(trace);if err!=nil{t.Fatal(err)}
  if !observed {originalTrace=string(raw);originalRows=rows} else if originalTrace!=string(raw) || originalRows!=rows {t.Fatal("observer changed complete public trace or all durable fields")}
  finish()
 }
 t.Logf("private manual/page observer off/on exact trace SHA256=%x; durable10tables identical; acceptancefalse",sha256.Sum256([]byte(originalTrace)))
}
